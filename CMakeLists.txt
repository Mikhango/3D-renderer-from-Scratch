cmake_minimum_required(VERSION 3.16)
project(3d-renderer VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Eigen3 REQUIRED NO_MODULE)
option(BUILD_GUI "Build the Qt desktop application" ON)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(renderer_core
    src/core/Camera.cpp src/core/Frame.cpp src/core/Lighting.cpp
    src/core/Mesh.cpp src/core/Model.cpp src/core/ObjLoader.cpp
    src/core/Primitives.cpp src/core/Rasterizer.cpp src/core/Renderer.cpp
    src/core/Scene.cpp src/core/ShadowScene.cpp src/core/ZBuffer.cpp)
target_include_directories(renderer_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(renderer_core PUBLIC Eigen3::Eigen)
if(MSVC)
    target_compile_options(renderer_core PRIVATE /W4)
else()
    target_compile_options(renderer_core PRIVATE -Wall -Wextra -Wpedantic)
    if(ENABLE_SANITIZERS)
        target_compile_options(renderer_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(renderer_core PUBLIC -fsanitize=address,undefined)
    endif()
endif()
if(BUILD_GUI)
    find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Gui Widgets)
    if(QT_VERSION_MAJOR EQUAL 5)
        find_package(Qt5 5.14 REQUIRED COMPONENTS Core Gui Widgets)
    else()
        find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
    endif()
    set(CMAKE_AUTOMOC ON)
    add_library(renderer_gui
        src/app/Application.cpp src/app/Except.cpp src/app/QRunTime.cpp
        src/gui/GuiController.cpp src/gui/InputController.cpp
        src/gui/RenderView.cpp src/gui/SettingsWindow.cpp)
    target_link_libraries(renderer_gui PUBLIC renderer_core Qt${QT_VERSION_MAJOR}::Widgets)
    add_executable(3d-renderer src/main.cpp)
    target_link_libraries(3d-renderer PRIVATE renderer_gui)
    set_target_properties(3d-renderer PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
    foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
        set_target_properties(3d-renderer PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} ${CMAKE_BINARY_DIR}/bin)
    endforeach()
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE renderer_core)
    add_test(NAME core_tests COMMAND core_tests)
    set_tests_properties(core_tests PROPERTIES TIMEOUT 30)
    if(BUILD_GUI)
        add_executable(gui_tests tests/gui_tests.cpp)
        target_link_libraries(gui_tests PRIVATE renderer_gui)
        add_test(NAME gui_tests COMMAND gui_tests)
        set_tests_properties(gui_tests PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
endif()
